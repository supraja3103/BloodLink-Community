  <script>
    const store = { donors: 'bloodlink_donors', requests: 'bloodlink_requests' };
    const read = (k,f)=>JSON.parse(localStorage.getItem(k)||JSON.stringify(f));
    const write = (k,v)=>localStorage.setItem(k,JSON.stringify(v));
    let donors = read(store.donors,[{name:'Rahul Kumar',age:'26',bloodGroup:'O+',phone:'9876543210',email:'rahul@example.com',location:'Madhapur',availability:'Evenings',lastDonation:'2026-03-12',notes:'Ready for emergencies',createdAt:Date.now()-5000000}]);
    let requests = read(store.requests,[{patientName:'Sita Devi',bloodGroup:'O+',urgency:'Critical',units:'2',hospital:'City Care Hospital',location:'Miyapur',reason:'Accident emergency',status:'Open',createdAt:Date.now()-4000000}]);

    const tabs = document.querySelectorAll('.tab');
    const panels = {
      home:document.getElementById('homePanel'),
      patient:document.getElementById('patientPanel'),
      donor:document.getElementById('donorPanel'),
      admin:document.getElementById('adminPanel')
    };

    const donorForm = document.getElementById('donorForm');
    const requestForm = document.getElementById('requestForm');
    const donorList = document.getElementById('donorList');
    const requestList = document.getElementById('requestList');
    const searchDonor = document.getElementById('searchDonor');
    const bloodFilter = document.getElementById('bloodFilter');

    const totalDonors = document.getElementById('totalDonors');
    const totalRequests = document.getElementById('totalRequests');
    const totalMatches = document.getElementById('totalMatches');
    const totalUnits = document.getElementById('totalUnits');

    const save = ()=>{
      write(store.donors,donors);
      write(store.requests,requests);
    };

    const renderStats = ()=>{
      totalDonors.textContent = donors.length;
      totalRequests.textContent = requests.length;
      totalMatches.textContent = requests.reduce((a,r)=>a+donors.filter(d=>d.bloodGroup===r.bloodGroup).length,0);
      totalUnits.textContent = requests.reduce((a,r)=>a+Number(r.units||0),0);
    };

    const donorItem = d => `
      <article class="item">
        <div class="item-top">
          <div>
            <h4>${d.name} <span style="color:var(--primary)">(${d.bloodGroup})</span></h4>
            <div class="meta">
              Age: ${d.age} • ${d.location}<br>
              Phone: ${d.phone} • Email: ${d.email}<br>
              Availability: ${d.availability} • Last Donation: ${d.lastDonation || 'N/A'}
            </div>
          </div>
          <button class="ok" onclick="alert('Donor selected: ${d.name}')">Contact</button>
        </div>
        <div class="chips">
          <span class="chip">${d.bloodGroup}</span>
          <span class="chip">${d.location}</span>
          <span class="chip">${d.availability}</span>
        </div>
        ${d.notes ? `<div class="small" style="margin-top:10px">${d.notes}</div>` : ''}
      </article>
    `;

    const requestItem = r => {
      const matches = donors.filter(d=>d.bloodGroup===r.bloodGroup).length;
      return `
        <article class="item">
          <div class="item-top">
            <div>
              <h4>${r.patientName} needs ${r.bloodGroup} • ${r.units} unit(s)</h4>
              <div class="meta">
                Hospital: ${r.hospital}<br>
                Location: ${r.location} • Urgency: ${r.urgency}<br>
                Reason: ${r.reason || 'Not provided'}
              </div>
            </div>
            <button class="ok ${matches ? '' : 'danger'}" onclick="alert('${matches ? matches+' matching donor(s) found' : 'No matching donor found'}')">
              ${matches ? `Match (${matches})` : 'No Match'}
            </button>
          </div>
          <div class="chips">
            <span class="chip">${r.bloodGroup}</span>
            <span class="chip">${r.urgency}</span>
            <span class="chip">${r.status}</span>
          </div>
        </article>
      `;
    };

    const renderDonors = ()=>{
      const q = searchDonor.value.toLowerCase().trim();
      let items = donors.filter(d=>
        `${d.name} ${d.bloodGroup} ${d.location}`.toLowerCase().includes(q) &&
        (bloodFilter.value === 'All' || d.bloodGroup === bloodFilter.value)
      );
      donorList.innerHTML = items.length ? items.map(donorItem).join('') : '<div class="empty">No donors found.</div>';
    };

    const renderRequests = ()=>{
      requestList.innerHTML = requests.length ? requests.map(requestItem).join('') : '<div class="empty">No active requests.</div>';
    };

    const renderAll = ()=>{
      save();
      renderStats();
      renderDonors();
      renderRequests();
    };

    tabs.forEach(t=>t.onclick=()=>{
      tabs.forEach(x=>x.classList.remove('active'));
      t.classList.add('active');
      Object.values(panels).forEach(p=>p.classList.remove('active'));
      panels[t.dataset.tab].classList.add('active');
    });

    donorForm.onsubmit = e => {
      e.preventDefault();
      const d = Object.fromEntries(new FormData(donorForm).entries());
      d.createdAt = Date.now();
      donors.unshift(d);
      donorForm.reset();
      renderAll();
      alert('Donor saved successfully');
    };

    requestForm.onsubmit = e => {
      e.preventDefault();
      const r = Object.fromEntries(new FormData(requestForm).entries());
      r.status='Open';
      r.createdAt = Date.now();
      requests.unshift(r);
      requestForm.reset();
      renderAll();
      alert('Request created successfully');
    };

    searchDonor.oninput = renderDonors;
    bloodFilter.onchange = renderDonors;

    document.getElementById('resetBtn').onclick = ()=>{
      localStorage.removeItem(store.donors);
      localStorage.removeItem(store.requests);
      location.reload();
    };

    document.getElementById('emergencyBtn').onclick = ()=>{
      alert('Emergency contact: 108 / nearest blood bank');
    };

    document.getElementById('seedBtn').onclick = ()=>{
      donors = [
        {name:'Rahul Kumar',age:'26',bloodGroup:'O+',phone:'9876543210',email:'rahul@example.com',location:'Madhapur',availability:'Evenings',lastDonation:'2026-03-12',notes:'Ready for emergencies',createdAt:Date.now()-5000000},
        {name:'Ananya Reddy',age:'24',bloodGroup:'A+',phone:'9123456780',email:'ananya@example.com',location:'Kukatpally',availability:'Weekends',lastDonation:'2026-02-20',notes:'Can travel nearby',createdAt:Date.now()-3000000},
        {name:'Sameer Khan',age:'31',bloodGroup:'B+',phone:'9012345678',email:'sameer@example.com',location:'Gachibowli',availability:'Anytime',lastDonation:'2025-12-15',notes:'Available for camp',createdAt:Date.now()-1000000}
      ];
      requests = [
        {patientName:'Sita Devi',bloodGroup:'O+',urgency:'Critical',units:'2',hospital:'City Care Hospital',location:'Miyapur',reason:'Accident emergency',status:'Open',createdAt:Date.now()-4000000},
        {patientName:'Arjun Verma',bloodGroup:'A+',urgency:'High',units:'1',hospital:'Apollo Clinic',location:'Hitech City',reason:'Surgery support',status:'Open',createdAt:Date.now()-2000000}
      ];
      renderAll();
    };

    document.getElementById('approveAllBtn').onclick = ()=>{
      requests = requests.map(r=>({...r,status:'Matched'}));
      renderAll();
    };

    document.getElementById('clearHistoryBtn').onclick = ()=>{
      requests = requests.filter(r=>r.status === 'Open');
      renderAll();
    };

    renderAll();
  </script>
